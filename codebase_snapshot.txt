// src/api/client.js
import axios from "axios";
import Constants from "expo-constants";
import * as SecureStore from "expo-secure-store";

const API_URL =
  Constants.expoConfig?.extra?.API_URL ||
  "https://skilledpro-backend-production.up.railway.app/api";

const ACCESS_KEY = "skp_access";
const REFRESH_KEY = "skp_refresh";

export const tokenStore = {
  async getAccess() {
    try { return await SecureStore.getItemAsync(ACCESS_KEY); } catch { return null; }
  },
  async getRefresh() {
    try { return await SecureStore.getItemAsync(REFRESH_KEY); } catch { return null; }
  },
  async setTokens(access, refresh) {
    try {
      if (access) await SecureStore.setItemAsync(ACCESS_KEY, access);
      if (refresh) await SecureStore.setItemAsync(REFRESH_KEY, refresh);
    } catch {}
  },
  async clear() {
    try {
      await SecureStore.deleteItemAsync(ACCESS_KEY);
      await SecureStore.deleteItemAsync(REFRESH_KEY);
    } catch {}
  },
};

const api = axios.create({
  baseURL: API_URL,
  timeout: 30000,
  headers: { "Content-Type": "application/json" },
});

// Attach access token
api.interceptors.request.use(async (config) => {
  const token = await tokenStore.getAccess();
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// Refresh on 401 once
let isRefreshing = false;
let queued = [];

const flush = (err, token) => {
  queued.forEach(({ resolve, reject }) => {
    if (err) reject(err);
    else resolve(token);
  });
  queued = [];
};

api.interceptors.response.use(
  (res) => res,
  async (error) => {
    const original = error.config;
    if (
      error.response?.status === 401 &&
      !original._retry &&
      !original.url?.includes("/auth/refresh") &&
      !original.url?.includes("/auth/login")
    ) {
      original._retry = true;
      const refresh = await tokenStore.getRefresh();
      if (!refresh) return Promise.reject(error);

      if (isRefreshing) {
        return new Promise((resolve, reject) => {
          queued.push({ resolve, reject });
        }).then((token) => {
          original.headers.Authorization = `Bearer ${token}`;
          return api(original);
        });
      }

      isRefreshing = true;
      try {
        const { data } = await axios.post(`${API_URL}/auth/refresh`, {
          refreshToken: refresh,
        });
        const newAccess = data?.data?.accessToken;
        const newRefresh = data?.data?.refreshToken;
        await tokenStore.setTokens(newAccess, newRefresh);
        flush(null, newAccess);
        original.headers.Authorization = `Bearer ${newAccess}`;
        return api(original);
      } catch (err) {
        flush(err, null);
        await tokenStore.clear();
        return Promise.reject(err);
      } finally {
        isRefreshing = false;
      }
    }
    return Promise.reject(error);
  },
);

/** Pull a friendly error message off an axios error. */
export function getErrorMessage(err, fallback = "Something went wrong") {
  return (
    err?.response?.data?.message ||
    err?.response?.data?.error ||
    err?.message ||
    fallback
  );
}

export default api;

// src/components/AppButton.js
import React from "react";
import {
  Pressable,
  ActivityIndicator,
  StyleSheet,
  View,
} from "react-native";
import { Ionicons } from "@expo/vector-icons";
import AppText from "./AppText";
import { useTheme } from "../contexts/ThemeContext";

/**
 * AppButton — primary CTA used everywhere.
 *
 * variant:
 *   - "primary" (default): solid orange, dark text — matches "Sign In", "Continue →"
 *   - "secondary": panel bg with border — matches "← Back"
 *   - "ghost": transparent
 *   - "danger": red
 *
 * size: "md" (default) | "sm" | "lg"
 */
export default function AppButton({
  title,
  onPress,
  variant = "primary",
  size = "md",
  loading = false,
  disabled = false,
  iconLeft,
  iconRight,
  fullWidth = true,
  style,
}) {
  const { theme } = useTheme();
  const palette = getPalette(variant, theme, disabled);
  const dims = SIZE[size];

  return (
    <Pressable
      onPress={onPress}
      disabled={disabled || loading}
      style={({ pressed }) => [
        styles.base,
        {
          backgroundColor: palette.bg,
          borderColor: palette.border,
          borderWidth: variant === "secondary" || variant === "ghost" ? 1 : 0,
          paddingVertical: dims.py,
          paddingHorizontal: dims.px,
          borderRadius: dims.radius,
          opacity: pressed ? 0.85 : 1,
          width: fullWidth ? "100%" : undefined,
          alignSelf: fullWidth ? "stretch" : "flex-start",
        },
        style,
      ]}
    >
      <View style={styles.row}>
        {loading ? (
          <ActivityIndicator color={palette.fg} />
        ) : (
          <>
            {iconLeft ? (
              <Ionicons
                name={iconLeft}
                size={dims.icon}
                color={palette.fg}
                style={{ marginRight: 8 }}
              />
            ) : null}
            <AppText
              variant={dims.textVariant}
              weight="semibold"
              style={{ color: palette.fg, textAlign: "center" }}
              numberOfLines={1}
            >
              {title}
            </AppText>
            {iconRight ? (
              <Ionicons
                name={iconRight}
                size={dims.icon}
                color={palette.fg}
                style={{ marginLeft: 8 }}
              />
            ) : null}
          </>
        )}
      </View>
    </Pressable>
  );
}

const SIZE = {
  sm: { py: 10, px: 16, radius: 10, icon: 16, textVariant: "small" },
  md: { py: 14, px: 20, radius: 12, icon: 18, textVariant: "subtitle" },
  lg: { py: 16, px: 24, radius: 14, icon: 20, textVariant: "subtitle" },
};

function getPalette(variant, theme, disabled) {
  const base = (() => {
    switch (variant) {
      case "secondary":
        return {
          bg: theme.colors.bgPanel,
          fg: theme.colors.text,
          border: theme.colors.border,
        };
      case "ghost":
        return {
          bg: "transparent",
          fg: theme.colors.text,
          border: theme.colors.border,
        };
      case "danger":
        return {
          bg: theme.colors.red,
          fg: "#ffffff",
          border: "transparent",
        };
      case "primary":
      default:
        return {
          // The website uses pure orange button with near-black text on the
          // "Sign In" CTA; we match that contrast.
          bg: theme.colors.orange,
          fg: "#0f0f1a",
          border: "transparent",
        };
    }
  })();

  if (disabled) {
    return { ...base, bg: theme.colors.orangeDim, fg: theme.colors.textDim };
  }
  return base;
}

const styles = StyleSheet.create({
  base: {
    alignItems: "center",
    justifyContent: "center",
  },
  row: {
    flexDirection: "row",
    alignItems: "center",
    justifyContent: "center",
  },
});

// src/components/AppCurrency.js
import React from "react";
import AppText from "./AppText";
import { useCurrency } from "../contexts/CurrencyContext";

/**
 * AppCurrency — render a monetary amount.
 *
 *   <AppCurrency amount={1500} />                     // uses global currency
 *   <AppCurrency amount={1500} currencyCode="NGN" />  // overrides for this instance
 *   <AppCurrency amount={1500} variant="title" />     // any AppText variant
 *
 * This is intentionally a thin component so screens never have to know how
 * formatting works for a given user's locale.
 */
export default function AppCurrency({
  amount,
  currencyCode,
  variant = "body",
  weight,
  tone,
  style,
  minimumFractionDigits,
  maximumFractionDigits,
}) {
  const { format } = useCurrency();
  const text = format(amount, {
    currencyCode,
    minimumFractionDigits,
    maximumFractionDigits,
  });

  return (
    <AppText variant={variant} weight={weight} tone={tone} style={style}>
      {text}
    </AppText>
  );
}

import React, { useState, forwardRef } from "react";
import {
  View,
  TextInput,
  Pressable,
  StyleSheet,
  I18nManager,
} from "react-native";
import { Ionicons } from "@expo/vector-icons";

import { useTheme } from "../contexts/ThemeContext";

/**
 * AppInput — themed input that matches the website's pill input
 *  (icon left, optional eye toggle for password, label above, error below).
 */
const AppInput = forwardRef(function AppInput(
  {
    label,
    iconLeft,
    placeholder,
    value,
    onChangeText,
    onBlur,
    onFocus,
    secureTextEntry,
    keyboardType = "default",
    autoCapitalize = "none",
    autoCorrect = false,
    autoComplete,
    textContentType,
    error,
    helperText,
    rightAction,
    rightActionLabel,
    onRightActionPress,
    multiline = false,
    numberOfLines,
    maxLength,
    editable = true,
    style,
  },
  ref,
) {
  const { theme } = useTheme();
  const [focused, setFocused] = useState(false);
  const [showPassword, setShowPassword] = useState(false);

  const isPassword = !!secureTextEntry;
  const obscured = isPassword && !showPassword;

  const borderColor = error
    ? theme.colors.red
    : focused
    ? theme.colors.borderFocus
    : theme.colors.border;

  return (
    <View style={[styles.wrap, style]}>
      {(label || onRightActionPress) && (
        <View style={styles.labelRow}>
          {label ? (
            <AppText variant="label" tone="dim">
              {label}
            </AppText>
          ) : (
            <View />
          )}
          {onRightActionPress ? (
            <Pressable onPress={onRightActionPress} hitSlop={6}>
              <AppText variant="small" tone="brand" weight="semibold">
                {rightActionLabel}
              </AppText>
            </Pressable>
          ) : null}
        </View>
      )}

      <View
        style={[
          styles.field,
          {
            backgroundColor: theme.colors.bgInput,
            borderColor,
            paddingVertical: multiline ? 12 : 0,
          },
        ]}
      >
        {iconLeft ? (
          <Ionicons
            name={iconLeft}
            size={18}
            color={theme.colors.textDim}
            style={styles.iconLeft}
          />
        ) : null}

        <TextInput
          ref={ref}
          value={value}
          onChangeText={onChangeText}
          placeholder={placeholder}
          placeholderTextColor={theme.colors.textMuted}
          onFocus={(e) => {
            setFocused(true);
            onFocus?.(e);
          }}
          onBlur={(e) => {
            setFocused(false);
            onBlur?.(e);
          }}
          secureTextEntry={obscured}
          keyboardType={keyboardType}
          autoCapitalize={autoCapitalize}
          autoCorrect={autoCorrect}
          autoComplete={autoComplete}
          textContentType={textContentType}
          multiline={multiline}
          numberOfLines={numberOfLines}
          maxLength={maxLength}
          editable={editable}
          maxFontSizeMultiplier={1.2}
          style={[
            styles.input,
            {
              color: theme.colors.text,
              textAlign: I18nManager.isRTL ? "right" : "left",
              fontFamily: "DMSans_400Regular",
              minHeight: multiline ? Math.max(48, (numberOfLines || 3) * 22) : 48,
              paddingLeft: iconLeft ? 0 : 14,
              paddingRight: isPassword ? 0 : 14,
            },
          ]}
        />

        {isPassword ? (
          <Pressable
            onPress={() => setShowPassword((s) => !s)}
            hitSlop={8}
            style={styles.eye}

              name={showPassword ? "eye-off-outline" : "eye-outline"}
              size={18}
              color={theme.colors.textDim}
            />
          </Pressable>
        ) : null}
      </View>

      {error ? (
        <AppText variant="small" tone="danger" style={styles.helper}>
          {error}
        </AppText>
      ) : helperText ? (
        <AppText variant="small" tone="muted" style={styles.helper}>
          {helperText}
        </AppText>
      ) : null}
    </View>
  );
});

// src/components/AppScreen.js
import React from "react";
import {
  View,
  StyleSheet,
  KeyboardAvoidingView,
  Platform,
  ScrollView,
  StatusBar as RNStatusBar,
} from "react-native";
import { SafeAreaView } from "react-native-safe-area-context";
import { StatusBar } from "expo-status-bar";
import { useTheme } from "../contexts/ThemeContext";

/**
 * AppScreen — every screen mounts inside this wrapper. It handles:
 *  - safe areas (notches, gesture bars)
 *  - themed background
 *  - keyboard avoidance
 *  - optional scroll
 *  - status bar style synced with the active theme
 *  - pull-to-refresh (pass a <RefreshControl /> via `refreshControl` prop)
 */
export default function AppScreen({
  children,
  scroll = true,
  keyboard = true,
  padded = true,
  edges = ["left", "right", "bottom"],
  contentContainerStyle,
  refreshControl,
  style,
}) {
  const { theme, effectiveMode } = useTheme();

  const body = scroll ? (
    <ScrollView
      keyboardShouldPersistTaps="handled"
      showsVerticalScrollIndicator={false}
      refreshControl={refreshControl}
      contentContainerStyle={[
        styles.scrollContent,
        padded && styles.padded,
        contentContainerStyle,
      ]}
    >
      {children}
    </ScrollView>
  ) : (
    <View
      style={[
        styles.viewContent,
        padded && styles.padded,
        contentContainerStyle,
      ]}
    >
      {children}
    </View>
  );

  return (
    <SafeAreaView
      edges={edges}
      style={[styles.safe, { backgroundColor: theme.colors.bg }, style]}
    >
      <StatusBar style={effectiveMode === "light" ? "dark" : "light"} />
      {keyboard ? (
        <KeyboardAvoidingView
          behavior={Platform.OS === "ios" ? "padding" : undefined}
          style={styles.flex}
          keyboardVerticalOffset={
            Platform.OS === "ios" ? 0 : RNStatusBar.currentHeight || 0
          }
        >
          {body}
        </KeyboardAvoidingView>
      ) : (
        body
      )}
    </SafeAreaView>
  );
}

const styles = StyleSheet.create({
  safe: { flex: 1 },
  flex: { flex: 1 },
  scrollContent: { flexGrow: 1 },
  viewContent: { flex: 1 },
  padded: { paddingHorizontal: 20, paddingVertical: 16 },
});

// src/components/AppText.js
import React from "react";
import { Text, StyleSheet } from "react-native";
import { useTheme } from "../contexts/ThemeContext";

/**
 * AppText — every piece of text in the app should use this component.
 *
 * Variants map to the website's typographic system:
 *   - "display" / "displaySm" → Syne (the bold "Sign in to SkilledProz" headlines)
 *   - "title"   → DM Sans 600, larger
 *   - "body"    → DM Sans 400 (default)
 *   - "label"   → uppercase tracking, used for form labels and "WELCOME BACK"
 *   - "muted"   → for hints and helper text
 *
 * `tone` overrides color: "default" | "dim" | "muted" | "brand" | "danger" | "success"
 *
 * Because text is sized in `sp` and we don't allow OS-level font scaling beyond a
 * cap, layouts hold up across long languages (German, Russian) without breaking.
 */
export default function AppText({
  children,
  variant = "body",
  tone = "default",
  weight,
  align,
  style,
  numberOfLines,
  selectable,
  ...rest
}) {
  const { theme, fonts } = useTheme();

  const variantStyle = VARIANTS[variant] || VARIANTS.body;
  const fontFamily = weight
    ? FAMILY_BY_WEIGHT[weight] || fonts.body
    : variantStyle.fontFamily(fonts);

  const color = TONE_COLOR(tone, theme);

  return (
    <Text
      maxFontSizeMultiplier={1.3}
      numberOfLines={numberOfLines}
      selectable={selectable}
      style={[
        styles.base,
        { fontFamily, color },
        variantStyle.style,
        align && { textAlign: align },
        style,
      ]}
      {...rest}
    >
      {children}
    </Text>
  );
}

const VARIANTS = {
  display: {
    fontFamily: (f) => f.displayBlack,
    style: { fontSize: 36, lineHeight: 42, letterSpacing: -0.5 },
  },
  displaySm: {
    fontFamily: (f) => f.display,
    style: { fontSize: 28, lineHeight: 34, letterSpacing: -0.3 },
  },
  title: {
    fontFamily: (f) => f.bodySemiBold,
    style: { fontSize: 20, lineHeight: 26 },
  },
  subtitle: {
    fontFamily: (f) => f.bodyMedium,
    style: { fontSize: 16, lineHeight: 22 },
  },
  body: {
    fontFamily: (f) => f.body,
    style: { fontSize: 15, lineHeight: 21 },
  },
  small: {
    fontFamily: (f) => f.body,
    style: { fontSize: 13, lineHeight: 18 },
  },
  label: {
    fontFamily: (f) => f.bodySemiBold,
    style: {
      fontSize: 12,
      lineHeight: 16,
      letterSpacing: 1.2,
      textTransform: "uppercase",
    },
  },
  muted: {
    fontFamily: (f) => f.body,
    style: { fontSize: 13, lineHeight: 18 },
  },
};

const FAMILY_BY_WEIGHT = {
  regular: "DMSans_400Regular",
  medium: "DMSans_500Medium",
  semibold: "DMSans_600SemiBold",
  bold: "DMSans_700Bold",
  display: "Syne_700Bold",
  displayBlack: "Syne_800ExtraBold",
};

const TONE_COLOR = (tone, theme) => {
  switch (tone) {
    case "dim":     return theme.colors.textDim;
    case "muted":   return theme.colors.textMuted;
    case "brand":   return theme.colors.orange;
    case "danger":  return theme.colors.red;
    case "success": return theme.colors.green;
    default:        return theme.colors.text;
  }
};

const styles = StyleSheet.create({
  base: {
    // RN's default includeFontPadding on Android adds extra vertical space —
    // disabling gives us pixel-accurate alignment with the web design.
    includeFontPadding: false,
  },
});

// src/components/Avatar.js
import React from "react";
import { View, Image, StyleSheet } from "react-native";
import AppText from "./AppText";
import { useTheme } from "../contexts/ThemeContext";

/**
 * Avatar — circular user photo with initials fallback.
 *
 *   <Avatar uri={user.avatar} firstName={user.firstName} lastName={user.lastName} size={40} />
 *
 * Falls back to a colored circle with the user's initials when no avatar.
 */
export default function Avatar({
  uri,
  firstName = "",
  lastName = "",
  size = 40,
  style,
}) {
  const { theme } = useTheme();
  const initials =
    `${firstName[0] || ""}${lastName[0] || ""}`.toUpperCase() || "?";

  if (uri) {
    return (
      <Image
        source={{ uri }}
        style={[
          styles.img,
          {
            width: size,
            height: size,
            borderRadius: size / 2,
            backgroundColor: theme.colors.bgInput,
          },
          style,
        ]}
      />
    );
  }

  return (
    <View
      style={[
        styles.fallback,
        {
          width: size,
          height: size,
          borderRadius: size / 2,
          backgroundColor: theme.colors.orangeDim,
          borderColor: theme.colors.border,

// src/components/Badge.js
import React from "react";
import { View, StyleSheet } from "react-native";
import AppText from "./AppText";
import { useTheme } from "../contexts/ThemeContext";

/**
 * Badge — themed status pill.
 *
 *   <Badge tone="success">Accepted</Badge>
 *   <Badge tone="danger">Disputed</Badge>
 *
 * Tones: "success" | "warning" | "danger" | "info" | "neutral"
 * Or pass `status` directly (PENDING / ACCEPTED / COMPLETED / etc.) and it
 * auto-picks the right tone.
 */
export default function Badge({ children, tone, status, size = "sm" }) {
  const { theme } = useTheme();
  const effectiveTone = tone || toneFromStatus(status);
  const palette = getPalette(effectiveTone, theme);
  const label = String(children ?? status ?? "").toUpperCase();

  return (
    <View
      style={[
        styles.base,
        size === "sm" ? styles.sm : styles.md,
        { backgroundColor: palette.bg, borderColor: palette.border },
      ]}
    >
      <AppText
        weight="bold"
        style={{
          color: palette.fg,
          fontSize: size === "sm" ? 10 : 11,
          letterSpacing: 0.8,
        }}
      >
        {label}
      </AppText>
    </View>
  );
}

function toneFromStatus(status) {
  switch ((status || "").toUpperCase()) {
    case "ACCEPTED":
    case "COMPLETED":
    case "RELEASED":
    case "VERIFIED":
    case "ACTIVE":
      return "success";
    case "PENDING":
    case "IN_PROGRESS":
    case "HELD":
    case "PROCESSING":
      return "warning";
    case "REJECTED":
    case "CANCELLED":
    case "DISPUTED":
    case "FAILED":
    case "REFUNDED":
      return "danger";
    case "OPEN":
      return "info";
    default:
      return "neutral";
  }
}

function getPalette(tone, theme) {
  switch (tone) {
    case "success":
      return {
        fg: theme.colors.green,
        bg: theme.colors.greenDim,
        border: theme.colors.greenDim,
      };
    case "warning":
      return {
        fg: theme.colors.orange,
        bg: theme.colors.orangeDim,
        border: theme.colors.orangeDim,
      };
    case "danger":
      return {
        fg: theme.colors.red,
        bg: theme.colors.redDim,
        border: theme.colors.redDim,
      };
    case "info":
      return {
        fg: theme.colors.orange,
        bg: theme.colors.orangeDim,
        border: theme.colors.orangeDim,
      };
    case "neutral":
    default:
      return {
        fg: theme.colors.textDim,
        bg: theme.colors.bgCard,
        border: theme.colors.border,
      };
  }
}

  },

/**
 * CurrencyPicker — pill button + modal sheet matching the website's
 * "Viewing in ₦ NGN ▾" control on the dashboard.
 *

export default function CurrencyPicker({ label }) {

  const { currency, setCurrencyCode } = useCurrency();
  const [open, setOpen] = useState(false);
  const [query, setQuery] = useState("");

  const filtered = useMemo(() => {
    if (!query.trim()) return CURRENCIES;
    const q = query.trim().toLowerCase();
    return CURRENCIES.filter(
      (c) =>
        c.code.toLowerCase().includes(q) ||
        c.name.toLowerCase().includes(q) ||
        c.symbol.toLowerCase().includes(q),
    );
  }, [query]);

  const handleSelect = async (code) => {
    await setCurrencyCode(code);
    setOpen(false);
    setQuery("");
  };

  return (
    <>
      <Pressable
        onPress={() => setOpen(true)}
        style={({ pressed }) => [
          styles.pill,
          {

            {label}
          </AppText>

      >
        <Pressable
          onPress={() => setOpen(false)}
          style={[styles.backdrop, { backgroundColor: "rgba(0,0,0,0.5)" }]}

          ]}
        >
          <View style={styles.header}>
            <AppText variant="title" weight="bold">
              {t("common.currency")}

            <Pressable onPress={() => setOpen(false)} hitSlop={10}>

/**
 * StatCard — matches the website dashboard stat tiles exactly.
 *
 *   ┌────────────────────┐
 *   │  📋                │   ← icon top-left in a rounded panel
 *   │                    │
 *   │                    │   ← flexible empty space
 *   │  64                │   ← big display number, colored when tone applied
 *   │  Total Bookings    │   ← muted label at bottom
 *   └────────────────────┘
 *
 * Props:
 *   icon  — emoji or short text rendered in the icon box
 *   label — the small caption (e.g. "Total Bookings")
 *   value — the big number; either a string/number or a React node
 *           (use a node for AppCurrency in the "Total Spent" tile)
 *   tone  — "neutral" | "warning" | "success"  — tints the background and value
 */
export default function StatCard({
  icon,
  label,
  value,
  tone = "neutral",
  onPress,
}) {
  const { theme } = useTheme();
  const palette = getPalette(tone, theme);
  const Wrap = onPress ? Pressable : View;

  return (
    <Wrap
      onPress={onPress}
      style={({ pressed } = {}) => [
        styles.card,
        {
          backgroundColor: palette.bg,
          borderColor: palette.border,
          opacity: pressed ? 0.85 : 1,
        },
      ]}
    >
      {/* Icon box top-left */}
      <View
        style={[
          styles.iconWrap,
          { backgroundColor: palette.iconBg, borderColor: palette.border },
        ]}
      >
        <AppText style={styles.icon}>{icon}</AppText>
      </View>

      {/* Big value */}
      <View style={styles.valueWrap}>
        {typeof value === "string" || typeof value === "number" ? (
          <AppText
            weight="displayBlack"
            style={[styles.value, { color: palette.valueColor }]}
            numberOfLines={1}
            adjustsFontSizeToFit
            minimumFontScale={0.5}
          >
            {value}
          </AppText>
        ) : (
          value
        )}

        <AppText
          variant="small"
          tone="dim"
          style={styles.label}
          numberOfLines={1}
        >
          {label}
        </AppText>
      </View>
    </Wrap>
  );
}

function getPalette(tone, theme) {
  switch (tone) {
    case "warning":
      return {
        bg: theme.colors.orangeDim,
        border: theme.colors.orange + "33",
        iconBg: theme.colors.bgPanel,
        valueColor: theme.colors.orange,
      };
    case "success":
      return {
        bg: theme.colors.greenDim,
        border: theme.colors.green + "33",
        iconBg: theme.colors.bgPanel,
        valueColor: theme.colors.green,
      };
    case "neutral":
    default:
      return {
        bg: theme.colors.bgPanel,
        border: theme.colors.border,
        iconBg: theme.colors.bgCard,
        valueColor: theme.colors.text,
      };
  }
}

import React, { useEffect, useState } from "react";
import { View, Pressable, StyleSheet, Platform } from "react-native";
import { Ionicons } from "@expo/vector-icons";
import { useNavigation, useNavigationState } from "@react-navigation/native";
import { useSafeAreaInsets } from "react-native-safe-area-context";

import AppText from "./AppText";
import Avatar from "./Avatar";
import { useTheme } from "../contexts/ThemeContext";
import { useAuthStore } from "../store/authStore";
import { hirerApi } from "../api/hirer";

/**
 * TopBar — sticky app header shown above every Hirer screen.
 *
 *   ┌──────────────────────────────────────────────────────────────┐
 *   │ ☰   Dashboard                              🔔 9+      [👤]   │
 *   └──────────────────────────────────────────────────────────────┘
 *
 *  - Hamburger opens the SideDrawer (via navigation.openDrawer())
 *  - Title is derived from the current route
 *  - Bell shows unread notification count badge
 *  - Avatar opens the profile screen
 *
 * Polls notifications every 30s so the badge stays fresh without sockets.
 */
export default function TopBar({ title: titleOverride }) {
  const { theme } = useTheme();
  const insets = useSafeAreaInsets();
  const navigation = useNavigation();
  const { user } = useAuthStore();
  const [unread, setUnread] = useState(0);

  // Derive the title from the current focused route name unless overridden.
  const routeName = useNavigationState(
    (state) => state?.routes?.[state.index]?.name,
  );
  const title = titleOverride || prettyTitle(routeName);

  useEffect(() => {
    let cancelled = false;
    let timer;
    const fetchUnread = async () => {
      try {
        const data = await hirerApi.getNotifications();
        if (!cancelled) setUnread(data?.unread ?? 0);
      } catch {
        // silent — TopBar must never crash a screen
      }
    };
    fetchUnread();
    timer = setInterval(fetchUnread, 30_000);
    return () => {
      cancelled = true;
      clearInterval(timer);
    };
  }, []);

  const openMenu = () => {
    // Works with the drawer navigator set up in HirerNavigator.
    if (navigation.openDrawer) {
      navigation.openDrawer();
    } else {
      navigation.getParent()?.openDrawer?.();
    }
  };

  const openNotifications = () => navigation.navigate("Notifications");
  const openProfile = () => navigation.navigate("Profile");

  return (
    <View
      style={[
        styles.wrap,
        {
          paddingTop: insets.top,
          backgroundColor: theme.colors.bg,
          borderBottomColor: theme.colors.border,
        },
      ]}
    >
      <View style={styles.row}>
        <Pressable
          onPress={openMenu}
          hitSlop={10}
          style={({ pressed }) => [
            styles.iconBtn,
            {
              backgroundColor: theme.colors.bgPanel,
              borderColor: theme.colors.border,
              opacity: pressed ? 0.7 : 1,
            },
          ]}
        >
          <Ionicons name="menu" size={20} color={theme.colors.text} />
        </Pressable>

        <AppText variant="title" weight="bold" style={styles.title}>
          {title}
        </AppText>

        <View style={styles.rightCluster}>
          <Pressable
            onPress={openNotifications}
            hitSlop={6}
            style={({ pressed }) => [
              styles.iconBtn,
              {
                backgroundColor: theme.colors.bgPanel,
                borderColor: theme.colors.border,
                opacity: pressed ? 0.7 : 1,
              },
            ]}
          >
            <Ionicons
              name="notifications-outline"
              size={20}
              color={theme.colors.text}
            />
            {unread > 0 ? (
              <View
                style={[styles.badge, { backgroundColor: theme.colors.orange }]}
              >
                <AppText
                  weight="bold"
                  style={{
                    color: "#0f0f1a",
                    fontSize: 10,
                    lineHeight: 12,
                  }}
                >
                  {unread > 9 ? "9+" : String(unread)}
                </AppText>
              </View>
            ) : null}
          </Pressable>

          <Pressable
            onPress={openProfile}
            hitSlop={6}
            style={({ pressed }) => [{ opacity: pressed ? 0.7 : 1 }]}
          >
            <Avatar
              uri={user?.avatar}
              firstName={user?.firstName}
              lastName={user?.lastName}
              size={40}
            />
          </Pressable>
        </View>
      </View>
    </View>
  );
}

function prettyTitle(route) {
  if (!route) return "Dashboard";
  // CamelCase → "Camel Case"
  return route.replace(/([A-Z])/g, " $1").trim();
}

});

export function CurrencyProvider({ children, initialCode = "USD" }) {
  const [code, setCode] = useState(initialCode);

  useEffect(() => {
    (async () => {
      try {
        const saved = await AsyncStorage.getItem(STORAGE_KEY);
        if (saved) setCode(saved);
      } catch {}
    })();
  }, []);

  const setCurrencyCode = async (next) => {
    setCode(next);
    try { await AsyncStorage.setItem(STORAGE_KEY, next); } catch {}
  };

  const currency = getCurrency(code);

  const format = useMemo(
    () => (amount, opts = {}) => {
      const targetCode = opts.currencyCode || code;
      const target = getCurrency(targetCode);
      const num = Number(amount ?? 0);
      try {
        return new Intl.NumberFormat(opts.locale || undefined, {
          style: "currency",
          currency: targetCode,
          minimumFractionDigits: opts.minimumFractionDigits ?? 0,
          maximumFractionDigits: opts.maximumFractionDigits ?? 2,
        }).format(num);
      } catch {
        const formatted = num.toLocaleString(undefined, {
          minimumFractionDigits: opts.minimumFractionDigits ?? 0,
          maximumFractionDigits: opts.maximumFractionDigits ?? 2,
        });
        return `${target.symbol}${formatted}`;
      }
    },
    [code]
  );

  const value = useMemo(
    () => ({ currency, currencyCode: code, setCurrencyCode, format, currencies: CURRENCIES }),
    [currency, code, format]
  );

  return <CurrencyContext.Provider value={value}>{children}</CurrencyContext.Provider>;
}

import React, {
  createContext,
  useContext,
  useState,
  useCallback,
  useRef,
  useEffect,
} from "react";
import {
  View,
  Text,
  StyleSheet,
  Animated,
  TouchableOpacity,
  Platform,
} from "react-native";
import { Ionicons } from "@expo/vector-icons";
import { useSafeAreaInsets } from "react-native-safe-area-context";
import { useTheme } from "./ThemeContext";

/**
 * Toast types: "success" | "error" | "info" | "warning"
 *
 * Usage anywhere in the tree:
 *   const { toast } = useToast();
 *   toast.success("Logged in");
 *   toast.error("Invalid credentials");
 *   toast.info("Heads up");
 *   toast.warning("Are you sure?");
 *
 * Always prefer this over Alert.alert.
 */

const ToastContext = createContext({
  show: () => {},
  toast: { success: () => {}, error: () => {}, info: () => {}, warning: () => {} },
});

const DURATION = 3500;

export function ToastProvider({ children }) {
  const [queue, setQueue] = useState([]);
  const idRef = useRef(0);

  const dismiss = useCallback((id) => {
    setQueue((q) => q.filter((t) => t.id !== id));
  }, []);

  const show = useCallback(
    (message, type = "info", opts = {}) => {
      const id = ++idRef.current;
      const toast = {
        id,
        message: String(message ?? ""),
        type,
        duration: opts.duration ?? DURATION,
      };
      setQueue((q) => [...q, toast]);
      return id;
    },
    [],
  );

  const toast = {
    success: (msg, opts) => show(msg, "success", opts),
    error: (msg, opts) => show(msg, "error", opts),
    info: (msg, opts) => show(msg, "info", opts),
    warning: (msg, opts) => show(msg, "warning", opts),
  };

  return (
    <ToastContext.Provider value={{ show, toast, dismiss }}>
      {children}
      <ToastViewport queue={queue} onDismiss={dismiss} />
    </ToastContext.Provider>
  );
}

function ToastViewport({ queue, onDismiss }) {
  const insets = useSafeAreaInsets();
  return (
    <View
      pointerEvents="box-none"
      style={[
        styles.viewport,
        { top: insets.top + 8 },
      ]}
    >
      {queue.map((t) => (
        <ToastItem key={t.id} toast={t} onDismiss={onDismiss} />
      ))}
    </View>
  );
}

function ToastItem({ toast, onDismiss }) {
  const { theme } = useTheme();
  const opacity = useRef(new Animated.Value(0)).current;
  const translateY = useRef(new Animated.Value(-12)).current;

  useEffect(() => {
    Animated.parallel([
      Animated.timing(opacity, {
        toValue: 1,
        duration: 220,
        useNativeDriver: true,
      }),
      Animated.timing(translateY, {
        toValue: 0,
        duration: 240,
        useNativeDriver: true,
      }),
    ]).start();

    const timer = setTimeout(handleDismiss, toast.duration);
    return () => clearTimeout(timer);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  const handleDismiss = () => {
    Animated.parallel([
      Animated.timing(opacity, {
        toValue: 0,
        duration: 180,
        useNativeDriver: true,
      }),
      Animated.timing(translateY, {
        toValue: -8,
        duration: 180,
        useNativeDriver: true,
      }),
    ]).start(() => onDismiss(toast.id));
  };

  const palette = getPalette(toast.type, theme);

  return (
    <Animated.View
      style={[
        styles.toast,
        {
          backgroundColor: theme.colors.bgPanel,
          borderColor: palette.border,
          shadowColor: theme.colors.shadow,
          opacity,
          transform: [{ translateY }],
        },
      ]}
    >
      <View
        style={[styles.iconWrap, { backgroundColor: palette.bgIcon }]}
      >
        <Ionicons name={palette.icon} size={18} color={palette.iconColor} />
      </View>
      <Text
        style={[styles.message, { color: theme.colors.text }]}
        numberOfLines={3}
      >
        {toast.message}
      </Text>
      <TouchableOpacity
        onPress={handleDismiss}
        hitSlop={{ top: 10, bottom: 10, left: 10, right: 10 }}
      >
        <Ionicons name="close" size={18} color={theme.colors.textDim} />
      </TouchableOpacity>
    </Animated.View>
  );
}

function getPalette(type, theme) {
  switch (type) {
    case "success":
      return {
        icon: "checkmark-circle",
        iconColor: theme.colors.green,
        bgIcon: theme.colors.greenDim,
        border: theme.colors.greenDim,
      };
    case "error":
      return {
        icon: "alert-circle",
        iconColor: theme.colors.red,
        bgIcon: theme.colors.redDim,
        border: theme.colors.redDim,
      };
    case "warning":
      return {
        icon: "warning",
        iconColor: theme.colors.orange,
        bgIcon: theme.colors.orangeDim,
        border: theme.colors.orangeDim,
      };
    case "info":
    default:
      return {
        icon: "information-circle",
        iconColor: theme.colors.orange,
        bgIcon: theme.colors.orangeDim,
        border: theme.colors.border,
      };
  }
}

const styles = StyleSheet.create({
  viewport: {
    position: "absolute",

    right: 12,
    zIndex: 9999,
    elevation: 9999,
    gap: 8,
  },

// src/navigation/RootNavigator.js
import React, { useEffect } from "react";
import { View, ActivityIndicator, StyleSheet } from "react-native";
import {
  NavigationContainer,
  DarkTheme,
  DefaultTheme,
} from "@react-navigation/native";
import { createNativeStackNavigator } from "@react-navigation/native-stack";

import AuthNavigator from "./AuthNavigator";
import HirerNavigator from "./HirerNavigator";
import VerifyEmailScreen from "../screens/auth/VerifyEmailScreen";
import { useAuthStore } from "../store/authStore";
import { useTheme } from "../contexts/ThemeContext";

const VerifyStack = createNativeStackNavigator();

/**
 * VerifyGate — a one-screen stack shown when the user is signed in but their
 * email is not yet verified. Wrapping it in a stack (rather than rendering the
 * screen bare) keeps useNavigation() working inside VerifyEmailScreen.
 */
function VerifyGate() {
  return (
    <VerifyStack.Navigator screenOptions={{ headerShown: false }}>
      <VerifyStack.Screen name="VerifyEmail" component={VerifyEmailScreen} />
    </VerifyStack.Navigator>
  );
}

export default function RootNavigator() {
  const { user, isHydrated, hydrate } = useAuthStore();
  const { theme, effectiveMode } = useTheme();

  useEffect(() => {
    hydrate();
  }, []);

  if (!isHydrated) {
    return (
      <View style={[styles.loading, { backgroundColor: theme.colors.bg }]}>
        <ActivityIndicator size="large" color={theme.colors.orange} />
      </View>
    );
  }

  // Build a navigation theme that matches our colors so back gestures and
  // page transitions don't show a white flash.
  const navTheme = {
    ...(effectiveMode === "light" ? DefaultTheme : DarkTheme),
    colors: {
      ...(effectiveMode === "light" ? DefaultTheme.colors : DarkTheme.colors),
      background: theme.colors.bg,
      card: theme.colors.bgPanel,
      text: theme.colors.text,
      border: theme.colors.border,
      primary: theme.colors.orange,
    },
  };

  // Three-way routing — no manual navigation needed. State drives which stack
  // renders. After register, user is set but isEmailVerified is false, so we
  // land on the VerifyGate automatically. Once /auth/me reports verified,
  // we swap into HirerNavigator with no extra navigation code.
  let stack;
  if (!user) {
    stack = <AuthNavigator />;
  } else if (!user.isEmailVerified) {
    stack = <VerifyGate />;
  } else {
    stack = <HirerNavigator />;
  }

  return <NavigationContainer theme={navTheme}>{stack}</NavigationContainer>;
}

const styles = StyleSheet.create({
  loading: { flex: 1, alignItems: "center", justifyContent: "center" },
});

import React, { useState } from "react";
import { View, Pressable, StyleSheet } from "react-native";
import { useTranslation } from "react-i18next";
import { useNavigation } from "@react-navigation/native";

import AppScreen from "../../components/AppScreen";
import AppText from "../../components/AppText";
import AppInput from "../../components/AppInput";
import AppButton from "../../components/AppButton";
import { useToast } from "../../contexts/ToastContext";
import { useAuthStore } from "../../store/authStore";
import { isEmail, isNonEmpty } from "../../utils/validators";

export default function LoginScreen() {
  const { t } = useTranslation();
  const navigation = useNavigation();
  const { toast } = useToast();
  const { login, isLoading } = useAuthStore();

  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [errors, setErrors] = useState({});

  const validate = () => {
    const e = {};
    if (!isNonEmpty(email)) e.email = t("auth.emailRequired");
    else if (!isEmail(email)) e.email = t("auth.validEmailRequired");
    if (!isNonEmpty(password)) e.password = t("auth.fillAllFields");
    setErrors(e);
    return Object.keys(e).length === 0;
  };

  const handleSubmit = async () => {
    if (!validate()) return;
    try {
      const user = await login(email.trim().toLowerCase(), password);
      toast.success(t("auth.loginSuccess"));
      // Navigation is driven by RootNavigator listening to auth state
      // No manual navigation needed.
    } catch (err) {
      toast.error(err.message || t("auth.invalidCredentials"));
    }
  };

  return (
    <AppScreen>
      <View style={styles.container}>
        <View style={styles.heroSpacer} />

        <AppText variant="label" tone="brand">
          {t("auth.welcomeBack")}
        </AppText>

        <AppText variant="display" weight="displayBlack" style={styles.title}>
          {t("auth.signInTitle")}
        </AppText>

        <View style={styles.subRow}>
          <AppText variant="body" tone="dim">
            {t("auth.noAccountYet")}{" "}
          </AppText>
          <Pressable onPress={() => navigation.navigate("Welcome")} hitSlop={6}>
            <AppText variant="body" tone="brand" weight="semibold">
              {t("auth.createOneFree")} →
            </AppText>
          </Pressable>
        </View>

        <View style={styles.form}>
          <AppInput
            label={t("auth.emailAddress")}
            iconLeft="mail-outline"
            placeholder={t("auth.emailPlaceholder")}
            value={email}
            onChangeText={setEmail}
            keyboardType="email-address"
            autoComplete="email"
            textContentType="emailAddress"
            error={errors.email}
          />

          <AppInput
            label={t("auth.password")}
            iconLeft="lock-closed-outline"
            placeholder={t("auth.passwordMinChars")}
            value={password}
            onChangeText={setPassword}
            secureTextEntry
            autoComplete="password"
            textContentType="password"
            rightActionLabel={t("auth.forgot")}
            onRightActionPress={() => navigation.navigate("ForgotPassword")}
            error={errors.password}
          />

          <AppButton
            title={isLoading ? t("auth.signingIn") : t("auth.signIn")}
            onPress={handleSubmit}
            loading={isLoading}
            style={styles.cta}
          />
        </View>
      </View>
    </AppScreen>
  );
}

  isEmail,
  isNonEmpty,
  isStrongEnoughPassword,
} from "../../utils/validators";

export default function RegisterScreen() {
  const { t } = useTranslation();
  const navigation = useNavigation();
  const route = useRoute();
  const { theme } = useTheme();
  const { toast } = useToast();
  const { register, isLoading } = useAuthStore();

  const role = route.params?.role || "HIRER";

  const [form, setForm] = useState({
    firstName: "",
    lastName: "",
    email: "",
    phone: "",
    country: "",
    city: "",
    password: "",
    confirm: "",
  });
  const [agree, setAgree] = useState(false);
  const [errors, setErrors] = useState({});

  const update = (key) => (value) => setForm((f) => ({ ...f, [key]: value }));

  const validate = () => {
    const e = {};
    if (!isNonEmpty(form.firstName)) e.firstName = t("auth.fillAllFields");
    if (!isNonEmpty(form.lastName)) e.lastName = t("auth.fillAllFields");
    if (!isNonEmpty(form.email)) e.email = t("auth.emailRequired");
    else if (!isEmail(form.email)) e.email = t("auth.validEmailRequired");
    if (!isStrongEnoughPassword(form.password))
      e.password = t("auth.passwordTooShort");

      toast.warning(t("auth.mustAgreeTerms"));
      return false;

          {t("auth.hirerAccount")}

          {t("auth.findNextProz")}
        </AppText>

        <View style={styles.form}>
          <View style={styles.row2}>
            <View style={styles.col}>
              <AppInput
                label={t("auth.firstName")}
                iconLeft="person-outline"
                placeholder={t("auth.firstNamePlaceholder")}
                value={form.firstName}
                onChangeText={update("firstName")}
                autoCapitalize="words"
                textContentType="givenName"
                error={errors.firstName}
              />
            </View>
            <View style={styles.col}>
              <AppInput

                autoCapitalize="words"
                textContentType="familyName"
                error={errors.lastName}
              />
            </View>
          </View>

          <AppInput
            label={t("auth.emailAddress")}
            iconLeft="mail-outline"
            placeholder={t("auth.emailPlaceholder")}
            value={form.email}
            onChangeText={update("email")}
            keyboardType="email-address"
            autoComplete="email"

// src/screens/auth/VerifyEmailScreen.js
import React, { useEffect, useRef, useState } from "react";
import { View, StyleSheet, AppState } from "react-native";
import { Ionicons } from "@expo/vector-icons";
import { useTranslation } from "react-i18next";

import AppScreen from "../../components/AppScreen";
import AppText from "../../components/AppText";
import AppButton from "../../components/AppButton";
import { useTheme } from "../../contexts/ThemeContext";
import { useToast } from "../../contexts/ToastContext";
import { useAuthStore } from "../../store/authStore";

/**
 * Mounted by RootNavigator when user is signed in but isEmailVerified === false.
 *
 *  - "Resend email" calls /auth/resend-verification
 *  - "I've verified" calls /auth/me; once the server flips the flag,
 *    RootNavigator auto-swaps to HirerNavigator (no navigation.navigate needed).
 *  - Also polls every 8s while the screen is focused, and fires fetchMe
 *    whenever the app returns to foreground (typical post-email-tap flow).
 */
export default function VerifyEmailScreen() {
  const { t } = useTranslation();
  const { theme } = useTheme();
  const { toast } = useToast();
  const { user, resendVerification, fetchMe, logout } = useAuthStore();

  const email = user?.email || "";
  const [sending, setSending] = useState(false);
  const [checking, setChecking] = useState(false);
  const pollRef = useRef(null);

  // Poll every 8s while mounted
  useEffect(() => {
    pollRef.current = setInterval(() => {
      fetchMe().catch(() => {});
    }, 8000);
    return () => clearInterval(pollRef.current);
  }, [fetchMe]);

  // Re-check immediately when the app comes back to foreground
  useEffect(() => {
    const sub = AppState.addEventListener("change", (state) => {
      if (state === "active") fetchMe().catch(() => {});
    });
    return () => sub.remove();
  }, [fetchMe]);

  const handleResend = async () => {
    if (!email) {
      toast.error(t("auth.emailRequired"));
      return;
    }
    setSending(true);
    try {
      await resendVerification(email);
      toast.success(t("auth.verificationResent"));
    } catch (err) {
      toast.error(err.message);
    } finally {
      setSending(false);
    }
  };

  const handleCheck = async () => {
    setChecking(true);
    try {
      const u = await fetchMe();
      if (!u?.isEmailVerified) {
        toast.info(t("auth.notVerifiedYet"));
      }
      // If verified, RootNavigator unmounts this screen automatically.
    } finally {
      setChecking(false);
    }
  };

  return (
    <AppScreen>
      <View style={styles.container}>
        <View style={{ height: 40 }} />

        <View
          style={[
            styles.iconBubble,
            { backgroundColor: theme.colors.orangeDim },
          ]}
        >
          <Ionicons
            name="mail-open-outline"
            size={36}
            color={theme.colors.orange}
          />
        </View>

        <AppText
          variant="displaySm"
          weight="displayBlack"
          style={{ marginTop: 24 }}
        >
          {t("auth.verifyEmailTitle")}
        </AppText>

        <AppText variant="subtitle" tone="dim" style={{ marginTop: 12 }}>
          {t("auth.verifyEmailBody", { email: email || "your inbox" })}
        </AppText>

        <View style={styles.actions}>
          <AppButton
            title={checking ? t("auth.checking") : t("auth.iVerified")}
            iconLeft={checking ? undefined : "checkmark-circle-outline"}
            onPress={handleCheck}
            loading={checking}
          />

          <AppButton
            title={sending ? t("auth.resending") : t("auth.resend")}
            variant="secondary"
            iconLeft={sending ? undefined : "refresh"}
            onPress={handleResend}
            loading={sending}
          />

          <AppButton
            title={t("auth.signOut")}
            variant="ghost"
            onPress={async () => {
              await logout();
              toast.info(t("auth.logoutSuccess"));
            }}
          />
        </View>
      </View>
    </AppScreen>
  );
}

const styles = StyleSheet.create({
  container: { gap: 8, alignItems: "flex-start" },

//   - 4 stat cards: Total Bookings / Active Jobs / Completed / Total Spent
//   - Recent Bookings list (5 items + View all)
//   - Recent Workers list (5 items + View all)
//
// Backend: GET /api/hirer/me/dashboard
export default function DashboardScreen() {
  const { t } = useTranslation();
  const navigation = useNavigation();
  const { theme } = useTheme();
  const { toast } = useToast();
  const { user } = useAuthStore();

  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);

  // Stable load function — empty dep array means same identity across renders,
  // which keeps the mount-effect below from re-firing in a loop. `t` and
  // `toast` are looked up via closure at call time, which is fine here
  // because we don't want to refetch on language or theme change anyway.
  const load = useCallback(async () => {
    try {
      const result = await hirerApi.getDashboard();
      setData(result);
    } catch (err) {
      toast.error(err?.response?.data?.message || t("toast.networkError"));
    } finally {
      setLoading(false);
      setRefreshing(false);
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  useEffect(() => {
    load();
  }, [load]);

  const onRefresh = useCallback(() => {
    setRefreshing(true);
    load();
  }, [load]);

    return (
      <AppScreen scroll={false}>
        <View style={styles.loadingWrap}>
          <ActivityIndicator size="large" color={theme.colors.orange} />
        </View>

// src/store/authStore.js
// Mirror of the website's `useAuthStore` adapted for React Native:
//   - tokens live in SecureStore (via tokenStore)
//   - user lives in AsyncStorage so we can rehydrate quickly on app open

import { create } from "zustand";
import AsyncStorage from "@react-native-async-storage/async-storage";
import api, { tokenStore, getErrorMessage } from "../api/client";

const USER_KEY = "@skilledproz:user";

export const useAuthStore = create((set, get) => ({
  user: null,
  isLoading: false,
  isHydrated: false,

  // Rehydrate user + decide auth state on app launch
  hydrate: async () => {
    try {
      const [rawUser, access] = await Promise.all([
        AsyncStorage.getItem(USER_KEY),
        tokenStore.getAccess(),
      ]);
      const cachedUser = rawUser ? JSON.parse(rawUser) : null;
      // Show cached user immediately for snappy UX
      set({ user: cachedUser });

      if (access) {
        // Validate by fetching /auth/me - this also picks up server-side updates
        try {
          const { data } = await api.get("/auth/me");
          set({ user: data.data });
          await AsyncStorage.setItem(USER_KEY, JSON.stringify(data.data));
        } catch {
          // Token invalid or refresh failed - clear
          await get().clearAuth();
        }
      }
    } finally {
      set({ isHydrated: true });
    }
  },

  setAuth: async (user, accessToken, refreshToken) => {
    await tokenStore.setTokens(accessToken, refreshToken);
    await AsyncStorage.setItem(USER_KEY, JSON.stringify(user));
    set({ user });
  },

  updateUser: async (patch) => {
    const next = { ...(get().user || {}), ...patch };
    set({ user: next });
    try {
      await AsyncStorage.setItem(USER_KEY, JSON.stringify(next));
    } catch {}
  },

  clearAuth: async () => {
    await tokenStore.clear();
    await AsyncStorage.removeItem(USER_KEY);
    set({ user: null });
  },

  fetchMe: async () => {
    try {
      const { data } = await api.get("/auth/me");
      set({ user: data.data });
      await AsyncStorage.setItem(USER_KEY, JSON.stringify(data.data));
      return data.data;
    } catch {
      await get().clearAuth();
      return null;
    }
  },

  login: async (email, password) => {
    set({ isLoading: true });
    try {
      const { data } = await api.post("/auth/login", { email, password });
      const { user, accessToken, refreshToken } = data.data;
      await get().setAuth(user, accessToken, refreshToken);
      return user;
    } catch (err) {
      throw new Error(getErrorMessage(err, "Login failed"));
    } finally {
      set({ isLoading: false });
    }
  },

  register: async (payload) => {
    set({ isLoading: true });
    try {
      const { data } = await api.post("/auth/register", payload);
      const { user, accessToken, refreshToken } = data.data;
      await get().setAuth(user, accessToken, refreshToken);
      return user;
    } catch (err) {
      throw new Error(getErrorMessage(err, "Registration failed"));
    } finally {
      set({ isLoading: false });
    }
  },

  logout: async () => {
    try {
      await api.post("/auth/logout");
    } catch {}
    await get().clearAuth();
  },

  forgotPassword: async (email) => {
    try {
      const { data } = await api.post("/auth/forgot-password", { email });
      return data;
    } catch (err) {
      throw new Error(getErrorMessage(err, "Could not send reset link"));
    }
  },

  resetPassword: async (token, password) => {
    try {
      const { data } = await api.post("/auth/reset-password", {
        token,
        password,
      });
      return data;
    } catch (err) {
      throw new Error(getErrorMessage(err, "Could not reset password"));
    }
  },

  resendVerification: async (email) => {
    try {
      const { data } = await api.post("/auth/resend-verification", { email });
      return data;
    } catch (err) {
      throw new Error(getErrorMessage(err, "Could not resend verification"));
    }
  },
}));
